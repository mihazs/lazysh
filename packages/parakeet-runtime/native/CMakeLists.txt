cmake_minimum_required(VERSION 3.20)
project(lazysh_parakeet_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # Match the actual pinned static ORT artifact, not the host compiler default.
  # x64 uses old libstdc++ strings; aarch64 uses the new ABI (verified symbols).
  # Mixed unversioned regex internals can abort during Ort::Env creation.
  if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    set(lazysh_ort_glibcxx_abi 0)
  elseif(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    set(lazysh_ort_glibcxx_abi 1)
  else()
    message(FATAL_ERROR "Pinned Linux ORT C++ ABI is verified for x86_64/aarch64 only")
  endif()
  add_compile_definitions($<$<COMPILE_LANGUAGE:CXX>:_GLIBCXX_USE_CXX11_ABI=${lazysh_ort_glibcxx_abi}>)
endif()
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_C_API ON CACHE BOOL "" FORCE)
foreach(feature BINARY PYTHON JNI PORTAUDIO WEBSOCKET TTS
    SPEAKER_DIARIZATION GPU DIRECTML TESTS CHECK RKNN AXERA AXCL ASCEND_NPU QNN SPACEMIT)
  set(SHERPA_ONNX_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(SHERPA_ONNX_BUILD_C_API_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_LINK_LIBSTDCPP_STATICALLY OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_USE_STATIC_CRT OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(ios_archive "${CMAKE_BINARY_DIR}/onnxruntime-ios.zip")
  file(DOWNLOAD "https://github.com/csukuangfj/onnxruntime-libs/releases/download/v1.28.2/onnxruntime-ios-static-xcframework-1.28.2.xcframework.zip"
    "${ios_archive}" EXPECTED_HASH "SHA256=2c2299acbb461d26d4bac4bc85985d40e7c7177ed6072703ae0846d88b0b4599" TLS_VERIFY ON)
  file(ARCHIVE_EXTRACT INPUT "${ios_archive}" DESTINATION "${CMAKE_BINARY_DIR}/ort-ios")
  if(LAZYSH_IOS_SIMULATOR)
    set(ios_slice "ios-arm64_x86_64-simulator")
  else()
    set(ios_slice "ios-arm64")
  endif()
  set(ios_framework "${CMAKE_BINARY_DIR}/ort-ios/onnxruntime.xcframework/${ios_slice}/onnxruntime.framework")
  set(ENV{SHERPA_ONNXRUNTIME_INCLUDE_DIR} "${ios_framework}/Headers")
  set(ENV{SHERPA_ONNXRUNTIME_LIB_DIR} "${CMAKE_BINARY_DIR}/ort-ios/onnxruntime.xcframework/${ios_slice}")
  set(SHERPA_ONNX_USE_PRE_INSTALLED_ONNXRUNTIME_IF_AVAILABLE ON CACHE BOOL "" FORCE)
else()
  set(SHERPA_ONNX_USE_PRE_INSTALLED_ONNXRUNTIME_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
endif()
set(revision "8abb7c8f87eae85b2e75f9a4115236c978980add")
set(source_hash "a5a7aac23fdc35fc4caf12e5a8ba63dc425ddb414904640bf13c4e7f38c3632b")
if(DEFINED ENV{LAZYSH_PARAKEET_SOURCE_ARCHIVE})
  set(archive "$ENV{LAZYSH_PARAKEET_SOURCE_ARCHIVE}")
else()
  set(archive "${CMAKE_BINARY_DIR}/sherpa-onnx.tar.gz")
  file(DOWNLOAD "https://codeload.github.com/k2-fsa/sherpa-onnx/tar.gz/${revision}"
    "${archive}" EXPECTED_HASH "SHA256=${source_hash}" TLS_VERIFY ON)
endif()
file(SHA256 "${archive}" actual_hash)
if(NOT actual_hash STREQUAL source_hash)
  message(FATAL_ERROR "Pinned sherpa-onnx source archive checksum mismatch")
endif()
set(prefix "sherpa-onnx-${revision}")
# Keep only native build inputs. Upstream scripts include an absolute symlink and are never extracted.
file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${CMAKE_BINARY_DIR}/vendor"
  PATTERNS "${prefix}/CMakeLists.txt" "${prefix}/LICENSE"
    "${prefix}/cmake/*" "${prefix}/sherpa-onnx/*" "${prefix}/toolchains/*")
set(upstream "${CMAKE_BINARY_DIR}/vendor/${prefix}")
add_subdirectory("${upstream}" upstream EXCLUDE_FROM_ALL)
add_library(lazysh-parakeet-native STATIC bridge.cc)
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  target_sources(lazysh-parakeet-native PRIVATE storage_macos.c)
endif()
target_include_directories(lazysh-parakeet-native PRIVATE "${upstream}")
target_link_libraries(lazysh-parakeet-native PRIVATE sherpa-onnx-c-api nlohmann_json::nlohmann_json)

set(native_libraries "$<TARGET_FILE:lazysh-parakeet-native>\n")
foreach(target sherpa-onnx-c-api sherpa-onnx-core kaldi-decoder-core kaldifst_core
    fstfar fst kaldi-native-fbank-core kissfft ssentencepiece_core utf8proc)
  string(APPEND native_libraries "$<TARGET_FILE:${target}>\n")
endforeach()
get_target_property(core_links sherpa-onnx-core LINK_LIBRARIES)
foreach(dependency IN LISTS core_links)
  if(IS_ABSOLUTE "${dependency}" AND dependency MATCHES "onnxruntime")
    string(APPEND native_libraries "${dependency}\n")
  elseif(dependency STREQUAL "onnxruntime")
    get_target_property(ort_archive onnxruntime IMPORTED_LOCATION)
    string(APPEND native_libraries "${ort_archive}\n")
  endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/native-libraries-$<CONFIG>.txt" CONTENT "${native_libraries}")
enable_testing()
add_executable(parakeet-boundary-tests EXCLUDE_FROM_ALL bridge.cc tests.cc)
target_include_directories(parakeet-boundary-tests PRIVATE "${upstream}")
target_link_libraries(parakeet-boundary-tests PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME exception-and-ownership-boundary COMMAND parakeet-boundary-tests)
set_tests_properties(exception-and-ownership-boundary PROPERTIES TIMEOUT 20)

if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  add_executable(parakeet-storage-tests EXCLUDE_FROM_ALL storage_macos.c storage_macos_tests.c)
  add_test(NAME macos-storage-boundary COMMAND parakeet-storage-tests)
  set_tests_properties(macos-storage-boundary PROPERTIES TIMEOUT 20)
endif()
if(CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin|Windows)$")
  add_executable(parakeet-runtime-tests EXCLUDE_FROM_ALL runtime_tests.cc)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(parakeet-runtime-tests PRIVATE LAZYSH_EXPECTED_GLIBCXX_ABI=${lazysh_ort_glibcxx_abi})
  endif()
  get_target_property(core_include_dirs sherpa-onnx-core INCLUDE_DIRECTORIES)
  target_include_directories(parakeet-runtime-tests PRIVATE ${core_include_dirs})
  target_link_libraries(parakeet-runtime-tests PRIVATE sherpa-onnx-core)
  add_test(NAME ort-environment-and-regex COMMAND parakeet-runtime-tests)
  set_tests_properties(ort-environment-and-regex PROPERTIES TIMEOUT 20)
endif()
