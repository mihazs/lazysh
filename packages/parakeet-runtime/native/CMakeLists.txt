# ORT's pinned Linux archive embeds an older protobuf than the system Mosh
# dependency. Hidden exports do not stop static Mosh relocations binding ORT's
# incompatible ExtensionSet, RTTI or vtables at the final client link.
function(lazysh_private_ort input output_variable)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(${output_variable} "${input}" PARENT_SCOPE)
    return()
  endif()
  if(NOT CMAKE_NM OR NOT CMAKE_OBJCOPY)
    message(FATAL_ERROR "The Linux ORT boundary requires target nm and objcopy")
  endif()
  execute_process(COMMAND "${CMAKE_NM}" --format=posix --extern-only "${input}"
    RESULT_VARIABLE nm_result OUTPUT_VARIABLE symbols ERROR_VARIABLE nm_error)
  if(NOT nm_result EQUAL 0)
    message(FATAL_ERROR "Cannot inspect the pinned ORT archive: ${nm_error}")
  endif()
  string(REPLACE "\n" ";" lines "${symbols}")
  set(protobuf_symbols "")
  foreach(line IN LISTS lines)
    string(REGEX MATCH "^[^ \t]+" symbol "${line}")
    if(symbol MATCHES "^_Z.*6google8protobuf")
      list(APPEND protobuf_symbols "${symbol}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES protobuf_symbols)
  list(SORT protobuf_symbols)
  list(LENGTH protobuf_symbols symbol_count)
  if(symbol_count EQUAL 0)
    message(FATAL_ERROR "Pinned Linux ORT has no expected protobuf namespace; refusing an unchecked ABI boundary")
  endif()
  set(mapping "")
  foreach(symbol IN LISTS protobuf_symbols)
    # Preserve valid Itanium mangling, including templates, RTTI, vtables and
    # undefined references; never rename STL, libc or the public Ort C API.
    string(REPLACE "6google8protobuf" "17lazysh_ort_google8protobuf" private_symbol "${symbol}")
    string(APPEND mapping "${symbol} ${private_symbol}\n")
  endforeach()
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/private-ort")
  set(map "${CMAKE_BINARY_DIR}/private-ort/protobuf-symbols.map")
  set(owned "${CMAKE_BINARY_DIR}/private-ort/libonnxruntime.a")
  file(WRITE "${map}" "${mapping}")
  # Only the owned build copy is transformed. The hash-verified publisher input
  # and all downloaded/source caches remain byte-for-byte untouched.
  configure_file("${input}" "${owned}" COPYONLY)
  execute_process(COMMAND "${CMAKE_OBJCOPY}" "--redefine-syms=${map}" "${owned}"
    RESULT_VARIABLE copy_result ERROR_VARIABLE copy_error)
  if(NOT copy_result EQUAL 0)
    message(FATAL_ERROR "Cannot isolate the pinned ORT protobuf archive: ${copy_error}")
  endif()
  execute_process(COMMAND "${CMAKE_NM}" --format=posix --extern-only "${owned}"
    RESULT_VARIABLE verify_result OUTPUT_VARIABLE isolated ERROR_VARIABLE verify_error)
  if(NOT verify_result EQUAL 0 OR isolated MATCHES "(^|\n)_Z[^ \t\n]*6google8protobuf"
      OR NOT isolated MATCHES "17lazysh_ort_google8protobuf"
      OR NOT isolated MATCHES "OrtGetApiBase")
    message(FATAL_ERROR "ORT private protobuf namespace verification failed: ${verify_error}")
  endif()
  message(STATUS "Isolated ${symbol_count} pinned ORT protobuf symbols")
  set(${output_variable} "${owned}" PARENT_SCOPE)
endfunction()

set(native_libraries "$<TARGET_FILE:lazysh-parakeet-native>\n")
foreach(target sherpa-onnx-c-api sherpa-onnx-core kaldi-decoder-core kaldifst_core
    fstfar fst kaldi-native-fbank-core kissfft ssentencepiece_core utf8proc)
  string(APPEND native_libraries "$<TARGET_FILE:${target}>\n")
endforeach()
get_target_property(core_links sherpa-onnx-core LINK_LIBRARIES)
set(private_core_links "")
set(ort_original "")
foreach(dependency IN LISTS core_links)
  if(IS_ABSOLUTE "${dependency}" AND dependency MATCHES "onnxruntime")
    set(ort_original "${dependency}")
    lazysh_private_ort("${dependency}" ort_archive)
    set(ort_private "${ort_archive}")
    string(APPEND native_libraries "${ort_archive}\n")
    list(APPEND private_core_links "${ort_archive}")
  elseif(dependency STREQUAL "onnxruntime")
    get_target_property(ort_archive onnxruntime IMPORTED_LOCATION)
    lazysh_private_ort("${ort_archive}" ort_archive)
    set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ort_archive}")
    string(APPEND native_libraries "${ort_archive}\n")
    list(APPEND private_core_links "${dependency}")
  else()
    list(APPEND private_core_links "${dependency}")
  endif()
endforeach()
set_property(TARGET sherpa-onnx-core PROPERTY LINK_LIBRARIES "${private_core_links}")
if(ort_original)
  # Upstream's unqualified target_link_libraries also publishes the original
  # path to consumers. Native CTest must exercise the same private archive as Rust.
  get_target_property(core_interface_links sherpa-onnx-core INTERFACE_LINK_LIBRARIES)
  if(core_interface_links)
    string(REPLACE "${ort_original}" "${ort_private}" core_interface_links "${core_interface_links}")
    set_property(TARGET sherpa-onnx-core PROPERTY INTERFACE_LINK_LIBRARIES "${core_interface_links}")
  endif()
endif()
