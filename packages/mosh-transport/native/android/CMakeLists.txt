cmake_minimum_required(VERSION 3.16)
project(lazysh_android_mosh_dependencies LANGUAGES C CXX)

# Pinned local sources only. No FetchContent, host libraries, or target protoc.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_BUILD_TEST_HELPERS OFF CACHE BOOL "" FORCE)
set(ABSL_ENABLE_INSTALL ON CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(protobuf_LOCAL_DEPENDENCIES_ONLY ON)
set(protobuf_BUILD_TESTS OFF)
set(protobuf_INSTALL ON)
set(protobuf_VERSION 36.2)
set(CMAKE_INSTALL_LIBDIR lib)
set(protobuf_SOURCE_DIR "${LAZYSH_VENDOR}/protobuf")
add_subdirectory("${LAZYSH_VENDOR}/abseil" abseil)
include("${protobuf_SOURCE_DIR}/cmake/abseil-cpp.cmake")
set(utf8_range_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(utf8_range_ENABLE_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${protobuf_SOURCE_DIR}/third_party/utf8_range" utf8_range)
include("${protobuf_SOURCE_DIR}/src/file_lists.cmake")
add_library(protobuf-lite STATIC ${libprotobuf_lite_srcs} ${libprotobuf_lite_hdrs})
target_include_directories(protobuf-lite PUBLIC "${protobuf_SOURCE_DIR}/src")
target_link_libraries(protobuf-lite PUBLIC ${protobuf_ABSL_USED_TARGETS} utf8_validity)
install(TARGETS protobuf-lite ARCHIVE DESTINATION lib)
